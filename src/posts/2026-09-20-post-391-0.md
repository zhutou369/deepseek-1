**程式碼示例：利用 DeepSeek API 進行關鍵字變體生成**

以下是一個使用 Python 呼叫 DeepSeek API 生成長尾關鍵字的範例：

```python
import os
import requests
import json

# 設定你的 DeepSeek API Key
DEEPSEEK_API_KEY = os.getenv("DEEPSEEK_API_KEY")
if not DEEPSEEK_API_KEY:
    raise ValueError("請設定 DEEPSEEK_API_KEY 環境變數")

DEEPSEEK_API_URL = "https://api.deepseek.com/chat/completions" # 假設這是聊天模型端點

def generate_long_tail_keywords(base_keyword: str, num_variations: int = 20):
    """
    使用 DeepSeek API 生成指定基礎關鍵字的長尾變體。
    """
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}"
    }

    # 構造請求的 messages 數組
    messages = [
        {"role": "system", "content": "你是一個資深的 SEO 專家和內容策略師，擅長挖掘長尾關鍵字。"},
        {"role": "user", "content": f"請為關鍵字「{base_keyword}」生成 {num_variations} 個相關的香港地區長尾關鍵字變體。這些關鍵字應具有高搜索意圖，且能用於創作具體內容。請以列表形式直接列出。"}
    ]

    payload = {
        "model": "deepseek-chat", # 或 deepseek-coder 等適合你需求的模型
        "messages": messages,
        "temperature": 0.7, # 控制生成內容的隨機性，0.7 是一個好的平衡點
        "max_tokens": 500
    }

    try:
        response = requests.post(DEEPSEEK_API_URL, headers=headers, data=json.dumps(payload))
        response.raise_for_status() # 如果請求失敗，拋出 HTTPError
        response_data = response.json()
        
        # 解析 DeepSeek 的回應
        if 'choices' in response_data and response_data['choices']:
            generated_text = response_data['choices'][0]['message']['content'].strip()
            # 簡單解析生成的列表，可能需要更複雜的正規表達式處理
            keywords = [line.strip() for line in generated_text.split('\n') if line.strip()]
            return keywords
        else:
            print("未能從 DeepSeek API 獲取有效的生成內容。")
            return []
    except requests.exceptions.RequestException as e:
        print(f"API 請求失敗: {e}")
        return []

if __name__ == "__main__":
    base_keyword_example = "香港自駕遊"
    generated_long_tails = generate_long_tail_keywords(base_keyword_example, num_variations=30)
    
    if generated_long_tails:
        print(f"為「{base_keyword_example}」生成的長尾關鍵字：")
        for i, kw in enumerate(generated_long_tails):
            print(f"{i+1}. {kw}")
    else:
        print("沒有生成任何長尾關鍵字。")
```

這個腳本能夠為你快速生成大量的長尾關鍵字，顯著提高關鍵字發現的效率。

3.  **細化要點：** 對於大綱中的每個部分，可以進一步要求 DeepSeek API 擴展其核心論點或包含的關鍵信息。

### 2. API 協同：規模化內容生成與優化

DeepSeek API 不僅能生成大綱，更能直接根據大綱和關鍵字生成初步的文章草稿。這使得內容生產能夠大規模進行。

**程式碼示例：利用 DeepSeek API 生成文章草稿**

```python
def generate_article_draft(theme_keyword: str, outline: list):
    """
    使用 DeepSeek API 根據主題關鍵字和文章大綱生成文章草稿。
    """
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}"
    }

    # 將大綱轉換為提示的一部分
    outline_str = "\n".join([f"- {item}" for item in outline])
    
    messages = [
        {"role": "system", "content": "你是一個專業的內容創作者和 SEO 文案，能根據大綱生成引人入勝且具備 SEO 友好性的文章。請使用香港繁體中文。"},
        {"role": "user", "content": f"請根據以下主題關鍵字「{theme_keyword}」和文章大綱，撰寫一篇內容豐富、結構清晰的文章草稿。請確保內容能滿足用戶的搜索意圖，並自然融入相關的長尾關鍵字。\n\n文章大綱：\n{outline_str}"}
    ]

    payload = {
        "model": "deepseek-chat",
        "messages": messages,
        "temperature": 0.8, # 稍微提高隨機性以獲得更豐富的內容
        "max_tokens": 2000 # 根據文章長度調整
    }

    try:
        response = requests.post(DEEPSEEK_API_URL, headers=headers, data=json.dumps(payload))
        response.raise_for_status()
        response_data = response.json()
        if 'choices' in response_data and response_data['choices']:
            return response_data['choices'][0]['message']['content'].strip()
        else:
            return "未能生成文章草稿。"
    except requests.exceptions.RequestException as e:
        return f"API 請求失敗: {e}"

if __name__ == "__main__":
    # 延續上一個示例的關鍵字
    article_theme = "香港自駕遊路線推薦"
    sample_outline = [
        "香港自駕遊的魅力與注意事項",
        "港島區經典自駕路線：石澳與赤柱",
        "新界區探索：大埔與西貢海鮮之旅",
        "租車指南與駕駛貼士",
        "自駕遊必備物品清單",
        "結語：規劃你的香港自駕探險"
    ]
    
    generated_draft = generate_article_draft(article_theme, sample_outline)
    print(f"\n生成的文章草稿：\n{generated_draft}")
```

生成草稿後，人工編輯仍是必不可少的環節。編輯人員可以專注於事實核查、語氣調整、品牌聲音融入和最終的 SEO 微調，極大地提升內容生產效率。
